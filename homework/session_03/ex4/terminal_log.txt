# Chạy song song hai trang web tĩnh trên cùng một IP, phân biệt bằng cổng.
# Cổng 80 vẫn do site "my-web" (file riêng trong sites-enabled) phục vụ, không thay đổi.

# --------------------------------------------------------------------------
# Server block 1: Beta App - cổng 8080
# --------------------------------------------------------------------------
server {
    # Lắng nghe IPv4 và IPv6 trên cổng 8080
    listen 8080;
    listen [::]:8080;

    # Virtual host theo cổng: không phân biệt tên miền nên dùng server_name "_"
    server_name _;

    # Thư mục mã nguồn của Beta App
    root /var/www/beta-app/html;
    index index.html;

    # Log riêng cho từng site để dễ truy vết
    access_log /var/log/nginx/beta-app.access.log;
    error_log  /var/log/nginx/beta-app.error.log;

    location / {
        # Trả file hoặc thư mục tương ứng, nếu không có thì trả 404
        try_files $uri $uri/ =404;
    }
}

# --------------------------------------------------------------------------
# Server block 2: Internal App - cổng 8090
# --------------------------------------------------------------------------
server {
    # Lắng nghe IPv4 và IPv6 trên cổng 8090
    listen 8090;
    listen [::]:8090;

    server_name _;

    # Thư mục mã nguồn của Internal App
    root /var/www/internal-app/html;
    index index.html;

    access_log /var/log/nginx/internal-app.access.log;
    error_log  /var/log/nginx/internal-app.error.log;

    location / {
        try_files $uri $uri/ =404;
    }
}
